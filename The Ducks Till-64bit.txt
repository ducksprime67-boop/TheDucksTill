def generate_valid_pdf(filename, text_lines):
    content_stream = ["BT", "/F1 10 Tf", "12 TL", "40 750 Td"]
    for line in text_lines:
        clean_line = line.replace("\\", "\\\\").replace("(", "\\(").replace(")", "\\)")
        content_stream.append(f"({clean_line}) '")
    content_stream.append("ET")
    
    stream_data = "\n".join(content_stream).encode("latin-1", "replace")
    stream_len = len(stream_data)

    pdf_objects = []
    pdf_objects.append("1 0 obj\n<< /Type /Catalog /Pages 2 0 R >>\nendobj\n")
    pdf_objects.append("2 0 obj\n<< /Type /Pages /Kids [3 0 R] /Count 1 >>\nendobj\n")
    pdf_objects.append("3 0 obj\n<< /Type /Page /Parent 2 0 R /Resources 4 0 R /MediaBox [0 0 595 842] /Contents 5 0 R >>\nendobj\n")
    pdf_objects.append("4 0 obj\n<< /Font << /F1 << /Type /Font /Subtype /Type1 /BaseFont /Courier >> >> >>\nendobj\n")
    pdf_objects.append(f"5 0 obj\n<< /Length {stream_len} >>\nstream\n".encode("latin-1") + stream_data + b"\nendstream\nendobj\n")

    xref_offsets = [0]
    output = bytearray()
    output.extend(b"%PDF-1.4\n")

    for obj in pdf_objects:
        xref_offsets.append(len(output))
        if isinstance(obj, str):
            output.extend(obj.encode("latin-1"))
        else:
            output.extend(obj)

    xref_start = len(output)
    output.extend(f"xref\n0 {len(xref_offsets)}\n0000000000 65535 f \n".encode("latin-1"))
    for offset in xref_offsets[1:]:
        output.extend(f"{offset:010d} 00000 n \n".encode("latin-1"))

    output.extend(f"trailer\n<< /Size {len(xref_offsets)} /Root 1 0 R >>\nstartxref\n{xref_start}\n%%EOF\n".encode("latin-1"))

    with open(filename, "wb") as f:
        f.write(output)

class Tooltip:
    def __init__(self, widget, text):
        self.widget = widget
        self.text = text
        self.tip_window = None
        self.widget.bind("<Enter>", self.show_tip)
        self.widget.bind("<Leave>", self.hide_tip)

    def show_tip(self, event=None):
        if self.tip_window or not self.text:
            return
        x = self.widget.winfo_rootx() + 20
        y = self.widget.winfo_rooty() + self.widget.winfo_height() + 5
        self.tip_window = tw = tk.Toplevel(self.widget)
        tw.wm_overrideredirect(True)
        tw.wm_geometry(f"+{x}+{y}")
        tw.attributes("-topmost", True)
        
        label = tk.Label(
            tw, text=self.text, justify="left",
            bg="#2A2A2A", fg="#FFD700", relief="solid", bd=1,
            font=("Segoe UI", 8, "bold"), padx=6, pady=4
        )
        label.pack()

    def hide_tip(self, event=None):
        if self.tip_window:
            self.tip_window.destroy()
            self.tip_window = None

    def global_arrow_up(self, event=None):
        widget = self.root.focus_get()
        if isinstance(widget, (tk.Entry, tk.Text)):
            return
        tree = self.get_active_treeview()
        if not tree:
            return
        children = tree.get_children()
        if not children:
            return "break"

        selection = tree.selection()
        if not selection:
            target = children[-1]
            tree.selection_set(target)
            tree.see(target)
        else:
            current = selection[-1]
            try:
                idx = children.index(current)
                if idx > 0:
                    target = children[idx - 1]
                else:
                    target = children[-1]
                tree.selection_set(target)
                tree.see(target)
            except Exception:
                pass
        return "break"

    def global_arrow_down(self, event=None):
        widget = self.root.focus_get()
        if isinstance(widget, (tk.Entry, tk.Text)):
            return
        tree = self.get_active_treeview()
        if not tree:
            return
        children = tree.get_children()
        if not children:
            return "break"

        selection = tree.selection()
        if not selection:
            target = children[0]
            tree.selection_set(target)
            tree.see(target)
        else:
            current = selection[-1]
            try:
                idx = children.index(current)
                if idx < len(children) - 1:
                    target = children[idx + 1]
                else:
                    target = children[0]
                tree.selection_set(target)
                tree.see(target)
            except Exception:
                pass
        return "break"

    def global_ctrl_a(self, event=None):
        widget = self.root.focus_get()
        if isinstance(widget, (tk.Entry, tk.Text)):
            return
        tree = self.get_active_treeview()
        if not tree:
            return "break"
        
        children = tree.get_children()
        if not children:
            return "break"

        current_selection = tree.selection()
        if len(current_selection) == len(children):
            tree.selection_remove(children)
        else:
            tree.selection_set(children)
        return "break"

    def render_image_on_canvas(self, canvas, src, max_w, max_h):
        canvas.delete("all")
        if not src:
            canvas.create_text(
                max_w / 2, max_h / 2, text="[ No Image Attached ]",
                fill=self.colors["muted"], font=("Segoe UI", 9, "bold")
            )
            return None
        
        cache_key = f"{src}_{max_w}_{max_h}"
        if cache_key in self.image_cache:
            photo = self.image_cache[cache_key]
            canvas.create_image(max_w / 2, max_h / 2, image=photo, anchor="center")
            return photo

        try:
            pil_img = None
            if src.startswith("http://") or src.startswith("https://"):
                req = Request(src, headers={"User-Agent": "Mozilla/5.0"})
                with urlopen(req, timeout=5) as resp:
                    pil_img = Image.open(io.BytesIO(resp.read()))
            elif os.path.exists(src):
                pil_img = Image.open(src)
            if pil_img:
                pil_img.thumbnail((max_w, max_h), Image.Resampling.LANCZOS)
                photo = ImageTk.PhotoImage(pil_img)
                self.image_cache[cache_key] = photo
                canvas.create_text(max_w / 2, max_h / 2, text="")
                canvas.create_image(max_w / 2, max_h / 2, image=photo, anchor="center")
                return photo
        except Exception:
            pass
        canvas.create_text(
            max_w / 2, max_h / 2, text="[ Invalid Image / URL ]",
            fill="#D9534F", font=("Segoe UI", 9, "bold")
        )
        return None

    def get_sales_count_map(self):
        sales_map = {}
        for s in self.data["sales"]:
            p_id = str(s.get("id", ""))
            sales_map[p_id] = sales_map.get(p_id, 0) + s.get("qty", 1)
        return sales_map

    def sort_products_list(self, products, sort_rule):
        sales_map = self.get_sales_count_map()
        prods = list(products)
        if sort_rule == "Best Selling (Highest Sold)":
            return sorted(prods, key=lambda p: sales_map.get(str(p.get("id", "")), 0), reverse=True)
        elif sort_rule == "Worst Selling (Lowest Sold)":
            return sorted(prods, key=lambda p: sales_map.get(str(p.get("id", "")), 0))
        elif sort_rule in ["Highest Price", "Highest Price (Rs)", "Highest Price ($)"]:
            return sorted(prods, key=lambda p: p.get("price", 0.0), reverse=True)
        elif sort_rule in ["Lowest Price", "Lowest Price (Rs)", "Lowest Price ($)"]:
            return sorted(prods, key=lambda p: p.get("price", 0.0))
        elif sort_rule == "Highest Stock Qty":
            return sorted(prods, key=lambda p: p.get("stock", 0), reverse=True)
        elif sort_rule == "Lowest Stock Qty":
            return sorted(prods, key=lambda p: p.get("stock", 0))
        return prods

    def sort_sales_list(self, sales, sort_rule):
        s_list = list(sales)
        if sort_rule in ["Highest Price", "Highest Price (Rs)", "Highest Price ($)"]:
            return sorted(s_list, key=lambda s: s.get("price", 0.0), reverse=True)
        elif sort_rule in ["Lowest Price", "Lowest Price (Rs)", "Lowest Price ($)"]:
            return sorted(s_list, key=lambda s: s.get("price", 0.0))
        elif sort_rule == "Best Selling (Highest Sold)":
            return sorted(s_list, key=lambda s: s.get("qty", 1), reverse=True)
        elif sort_rule == "Worst Selling (Lowest Sold)":
            return sorted(s_list, key=lambda s: s.get("qty", 1))
        return s_list

    def bind_tree_deselect_and_hover(self, tree, data_source_type):
        def on_click(event):
            region = tree.identify("region", event.x, event.y)
            item_id = tree.identify_row(event.y)
            
            if region == "heading":
                return

            if region == "nothing" or not item_id:
                tree.selection_remove(tree.selection())
                return

            if item_id:
                current_selection = tree.selection()
                if len(current_selection) == 1 and current_selection[0] == item_id:
                    tree.selection_remove(item_id)
                else:
                    tree.selection_set(item_id)
                    tree.focus(item_id)

        def on_mouse_move(event):
            region = tree.identify("region", event.x, event.y)
            if region == "heading":
                self.hide_hover_preview()
                return

            item_id = tree.identify_row(event.y)
            if not item_id:
                self.hide_hover_preview()
                return

            if item_id == self.last_hover_item:
                if self.hover_window:
                    try:
                        self.hover_window.wm_geometry(f"+{event.x_root + 15}+{event.y_root + 15}")
                    except Exception:
                        pass
                return
            
            self.last_hover_item = item_id
            vals = tree.item(item_id, "values")
            if not vals:
                self.hide_hover_preview()
                return

            img_src = ""
            p_name = ""
            p_category = ""
            
            if data_source_type == "inventory":
                target_id = str(vals[0])
                prod = next((p for p in self.data["inventory"] if str(p.get("id", "")) == target_id), None)
                if prod:
                    img_src = prod.get("image", "")
                    p_name = prod.get("name", "")
                    p_category = prod.get("category", "")
            elif data_source_type in ["sales", "return"]:
                target_sale = next((s for s in reversed(self.data["sales"]) if s["name"] == vals[0] and s["timestamp"] == vals[3]), None)
                if target_sale:
                    img_src = target_sale.get("image", "")
                    p_name = target_sale.get("raw_name", target_sale.get("name", ""))
                    p_category = target_sale.get("category", "")

            if self.hover_thread and self.hover_thread.is_alive():
                pass
            
            self.hover_thread = threading.Thread(target=self.async_load_and_show_preview, args=(event, img_src, p_name, p_category), daemon=True)
            self.hover_thread.start()

        def on_mouse_leave(event):
            self.last_hover_item = None
            self.hide_hover_preview()

        tree.bind("<Button-1>", on_click, add="+")
        tree.bind("<Motion>", on_mouse_move)
        tree.bind("<Leave>", on_mouse_leave)

    def async_load_and_show_preview(self, event, img_src, p_name, p_category):
        photo = None
        if img_src:
            cache_key = f"hover_opt_{img_src}"
            if cache_key in self.image_cache:
                photo = self.image_cache[cache_key]
            else:
                try:
                    pil_img = None
                    if img_src.startswith("http://") or img_src.startswith("https://"):
                        req = Request(img_src, headers={"User-Agent": "Mozilla/5.0"})
                        with urlopen(req, timeout=3) as resp:
                            pil_img = Image.open(io.BytesIO(resp.read()))
                    elif os.path.exists(img_src):
                        pil_img = Image.open(img_src)
                    
                    if pil_img:
                        pil_img.thumbnail((110, 110), Image.Resampling.LANCZOS)
                        photo = ImageTk.PhotoImage(pil_img)
                        self.image_cache[cache_key] = photo
                except Exception:
                    pass

        self.root.after(0, lambda: self.show_hover_preview_window(event, photo, p_name, p_category))

    def update_inventory_totals(self):
        sym, rate = self.get_current_currency_conversion()
        total_items = len(self.data["inventory"])
        total_qty = 0
        sum_of_unit_prices = 0.0
        grand_total_val = 0.0

        for p in self.data["inventory"]:
            qty = p.get("stock", 0)
            price = p.get("price", 0.0) * rate
            sum_of_unit_prices += price
            if qty < INFINITY_VAL:
                total_qty += qty
                grand_total_val += qty * price
            else:
                grand_total_val += price

        selected_items = self.inv_tree.selection()
        sel_count = len(selected_items)
        sel_val = 0.0

        for item_id in selected_items:
            vals = self.inv_tree.item(item_id, "values")
            if vals:
                p_id = str(vals[0])
                target = next((p for p in self.data["inventory"] if str(p.get("id", "")) == p_id), None)
                if target:
                    price = target.get("price", 0.0) * rate
                    qty = target.get("stock", 0)
                    sel_val += price * (qty if qty < INFINITY_VAL else 1)

        txt = f"📦 Products: {total_items} | Total Stock Qty: {total_qty} | Total: {sym} {sum_of_unit_prices:,.2f} | Grand Total: {sym} {grand_total_val:,.2f}"
        if sel_count > 0:
            txt += f"  ||  📌 Selected: {sel_count} item(s) ({sym} {sel_val:,.2f})"

        if hasattr(self, "inv_totals_summary_lbl"):
            self.inv_totals_summary_lbl.config(text=txt)

    def add_product(self):
        name = self.prod_name_entry.get().strip()
        category = self.category_dropdown.var.get()
        price = self.prod_price_entry.get().strip()
        stock = self.prod_stock_entry.get().strip()

        if not name or not price or not stock:
            messagebox.showerror("Error", "Please provide product name, price, and stock quantity.")
            return

        try:
            price_float = float(price)
            stock_int = int(stock)
        except ValueError:
            messagebox.showerror("Error", "Price and stock must be valid numbers.")
            return

        # Store base price normalized in USD baseline
        _, rate = self.get_current_currency_conversion()
        base_usd_price = price_float / rate

        prod_id = datetime.now().strftime("%Y%m%d%H%M%S")
        product = {
            "id": prod_id, "name": name, "category": category,
            "price": base_usd_price, "stock": stock_int,
            "image": self.current_image_source if self.current_image_source else ""
        }
        self.data["inventory"].append(product)
        save_data(self.data)

        self.prod_name_entry.delete(0, tk.END)
        self.prod_price_entry.delete(0, tk.END)
        self.prod_stock_entry.delete(0, tk.END)
        self.prod_stock_entry.insert(0, "10")
        self.prod_url_entry.delete(0, tk.END)
        self.current_image_source = None
        self.add_preview_canvas.delete("all")
        self.add_preview_canvas.create_text(
            100, 70, text="Drop Image / Browse\nor Paste Image Link",
            fill=self.colors["muted"], font=("Segoe UI", 9, "bold"), justify="center"
        )
        self.refresh_inventory_list()
        self.refresh_sale_list()
        self.refresh_return_list()
        messagebox.showinfo("Saved", f"'{name}' added permanently to inventory database!")

    def refresh_inventory_list(self, *args):
        sym, rate = self.get_current_currency_conversion()
        if hasattr(self, "price_label_ref"):
            self.price_label_ref.config(text=f"Price ({sym}):")

        for row in self.inv_tree.get_children():
            self.inv_tree.delete(row)
        query = self.inv_search_var.get().lower().strip()
        filter_low = self.low_stock_filter_var.get()
        sort_rule = self.inv_sort_dropdown.var.get() if hasattr(self, "inv_sort_dropdown") else "Default"

        filtered_products = []
        for p in self.data["inventory"]:
            stock_qty = p.get("stock", 0)
            if filter_low and stock_qty > 3:
                continue
            conv_price = p["price"] * rate
            if not query or query in p["name"].lower() or query in p["category"].lower() or query in str(conv_price):
                filtered_products.append(p)

        sorted_prods = self.sort_products_list(filtered_products, sort_rule)

        for p in sorted_prods:
            p_id = str(p.get("id", ""))
            stock_qty = p.get("stock", 0)
            price_val = p.get("price", 0.0) * rate

            if stock_qty >= INFINITY_VAL:
                tot_str = "∞"
            else:
                tot_str = f"{sym} {price_val * stock_qty:,.2f}"

            price_total_display = f"{sym} {price_val:,.2f}  |  Tot: {tot_str}"
            display_stock = "∞" if stock_qty >= INFINITY_VAL else str(stock_qty)
            
            row_values = [p_id, p["name"], p["category"], price_total_display, display_stock]
            self.inv_tree.insert("", "end", values=row_values)

        self.update_inventory_totals()

    def add_selected_to_cart(self):
        selected = self.sale_tree.selection()
        if not selected:
            messagebox.showwarning("Warning", "Select item(s) from catalog to add to cart.")
            return

        try:
            qty_val = int(self.pos_qty_entry.get().strip())
            if qty_val <= 0:
                raise ValueError()
        except ValueError:
            messagebox.showerror("Error", "Please enter a valid positive quantity number.")
            return

        try:
            custom_price_val = float(self.pos_price_entry.get().strip())
        except ValueError:
            messagebox.showerror("Error", "Please enter a valid final price.")
            return

        _, rate = self.get_current_currency_conversion()
        base_usd_custom_price = custom_price_val / rate

        for item_id in selected:
            vals = self.sale_tree.item(item_id, "values")
            if not vals:
                continue
            
            p_id = str(vals[0])
            product = next((p for p in self.data["inventory"] if str(p.get("id", "")) == p_id), None)
            if not product:
                product = next((p for p in self.data["inventory"] if p["name"] == vals[1]), None)
            
            if not product:
                continue

            existing = next((item for item in self.shopping_cart if item["id"] == str(product.get("id", "")) and item["price"] == base_usd_custom_price), None)
            if existing:
                existing["qty"] += qty_val
            else:
                self.shopping_cart.append({
                    "id": str(product.get("id", "")),
                    "name": product["name"],
                    "category": product["category"],
                    "price": base_usd_custom_price,
                    "qty": qty_val,
                    "image": product.get("image", "")
                })

        self.refresh_cart_display()
        self.pos_qty_entry.delete(0, tk.END)
        self.pos_qty_entry.insert(0, "1")

    def edit_cart_item_qty(self, event=None):
        selected = self.cart_tree.selection()
        if not selected:
            return
        vals = self.cart_tree.item(selected[0], "values")
        if not vals:
            return
        
        item_name = vals[0]
        cart_item = next((item for item in self.shopping_cart if item["name"] == item_name), None)
        if not cart_item:
            return

        new_qty_str = simpledialog.askstring("Edit Quantity", f"Enter new quantity for '{item_name}':", initialvalue=str(cart_item["qty"]), parent=self.root)
        if new_qty_str is not None:
            try:
                new_qty = int(new_qty_str)
                if new_qty > 0:
                    cart_item["qty"] = new_qty
                    self.refresh_cart_display()
                else:
                    self.shopping_cart = [item for item in self.shopping_cart if item["name"] != item_name]
                    self.refresh_cart_display()
            except ValueError:
                messagebox.showerror("Error", "Please enter a valid numeric quantity.", parent=self.root)

    def remove_cart_item(self):
        selected = self.cart_tree.selection()
        if not selected:
            messagebox.showwarning("Warning", "Select item from cart to remove.")
            return
        
        for item_id in selected:
            vals = self.cart_tree.item(item_id, "values")
            name_in_cart = vals[0]
            self.shopping_cart = [item for item in self.shopping_cart if item["name"] != name_in_cart]

        self.refresh_cart_display()

    def clear_shopping_cart(self):
        self.shopping_cart = []
        self.refresh_cart_display()

    def refresh_cart_display(self):
        sym, rate = self.get_current_currency_conversion()
        for row in self.cart_tree.get_children():
            self.cart_tree.delete(row)

        total_items = 0
        grand_total = 0.0

        for item in self.shopping_cart:
            qty = item["qty"]
            sub = (item["price"] * rate) * qty
            total_items += qty
            grand_total += sub
            self.cart_tree.insert("", "end", values=(item["name"], item["category"], qty, f"{sym} {sub:,.2f}"))

        self.cart_summary_lbl.config(text=f"🛒 Items: {total_items} | Grand Total: {sym} {grand_total:,.2f}")

    def complete_cart_checkout(self):
        if not self.shopping_cart:
            messagebox.showwarning("Warning", "Shopping cart is empty.")
            return

        sym, rate = self.get_current_currency_conversion()
        
        for item in self.shopping_cart:
            prod = next((p for p in self.data["inventory"] if str(p.get("id", "")) == str(item["id"])), None)
            if prod and prod.get("stock", 0) < INFINITY_VAL and item["qty"] > prod["stock"]:
                messagebox.showerror("Stock Error", f"Insufficient stock for '{item['name']}'! Available: {prod['stock']}")
                return

        timestamp = datetime.now().strftime("%Y-%m-%d %I:%M:%S %p")
        total_checkout_price = 0.0

        for item in self.shopping_cart:
            prod = next((p for p in self.data["inventory"] if str(p.get("id", "")) == str(item["id"])), None)
            if prod and prod.get("stock", 0) < INFINITY_VAL:
                prod["stock"] -= item["qty"]

            sub = (item["price"] * rate) * item["qty"]
            total_checkout_price += sub

            sale_record = {
                "id": item["id"],
                "name": f"{item['name']} (x{item['qty']})",
                "category": item["category"],
                "price": sub,
                "timestamp": timestamp,
                "image": item.get("image", ""),
                "qty": item["qty"],
                "raw_name": item["name"],
                "unit_price": item["price"] * rate
            }
            self.data["sales"].append(sale_record)

        self.shopping_cart = []
        save_data(self.data)

        self.refresh_cart_display()
        self.refresh_inventory_list()
        self.refresh_sale_list()
        self.refresh_report_list()
        self.refresh_return_list()
        self.populate_sales_checklist()

        messagebox.showinfo("Checkout Complete", f"Successfully completed cart checkout!\nTotal Paid: {sym} {total_checkout_price:,.2f}")

        selected = self.rep_tree.selection()

        for s in reversed(self.data["sales"]):

    def remove_sales_log(self):
        selected = self.rep_tree.selection()
        if not selected:
            messagebox.showwarning("Warning", "Select sales record(s) to delete.")
            return

        if not self.data["settings"].get("allow_staff_delete_sales", True):
            pin = simpledialog.askstring("Admin Verification", "Enter Admin PIN to delete sale log(s):", show="*")
            if pin != self.data["settings"].get("admin_pin", "1234"):
                messagebox.showerror("Access Denied", "Incorrect Admin PIN!")
                return

        confirm = self.data["settings"].get("confirm_actions", True)
        if confirm and not messagebox.askyesno("Confirm Delete", f"Are you sure you want to delete the {len(selected)} selected sales entry(ies)?"):
            return

        to_remove = set()
        for item_id in selected:
            vals = self.rep_tree.item(item_id, "values")
            to_remove.add((vals[0], vals[3]))

        self.data["sales"] = [s for s in self.data["sales"] if (s["name"], s["timestamp"]) not in to_remove]
        save_data(self.data)
        self.refresh_report_list()
        self.refresh_return_list()
        self.populate_sales_checklist()
        messagebox.showinfo("Deleted", "Selected sales log entry(ies) removed.")

    def export_sales_csv(self):
        if not self.data["sales"]:
            messagebox.showwarning("Warning", "No sales records to export.")
            return
        file_path = filedialog.asksaveasfilename(defaultextension=".csv", filetypes=[("CSV Files", "*.csv")], title="Export Sales Log")
        if file_path:
            try:
                sym, _ = self.get_current_currency_conversion()
                with open(file_path, "w", newline="", encoding="utf-8") as f:
                    writer = csv.writer(f)
                    writer.writerow(["Product Name", "Category", f"Sold Price ({sym})", "Timestamp"])
                    for s in self.data["sales"]:
                        writer.writerow([s["name"], s["category"], s["price"], s["timestamp"]])
                messagebox.showinfo("Exported", f"Saved sales log to:\n{file_path}")
            except Exception as e:
                messagebox.showerror("Error", str(e))

    def refresh_report_list(self, *args):
        sym, _ = self.get_current_currency_conversion()
        for row in self.rep_tree.get_children():
            self.rep_tree.delete(row)
        query = self.log_search_var.get().lower().strip()
        sort_rule = self.log_sort_dropdown.var.get() if hasattr(self, "log_sort_dropdown") else "Default"

        filtered_sales = []
        grand_total, sum_unit_prices, total_qty, count = 0.0, 0.0, 0, 0

        for s in reversed(self.data["sales"]):
            actual_price = s.get("price", 0.0)
            qty = s.get("qty", 1)
            unit_p = s.get("unit_price", actual_price / max(1, qty))
            
            grand_total += actual_price
            sum_unit_prices += unit_p
            total_qty += qty
            count += 1
            if not query or query in s["name"].lower() or query in s["category"].lower() or query in str(actual_price) or query in s["timestamp"].lower():
                filtered_sales.append(s)

        sorted_sales = self.sort_sales_list(filtered_sales, sort_rule)

        for s in sorted_sales:
            row_values = [s["name"], s["category"], f"{sym} {s['price']:.2f}", s["timestamp"]]
            self.rep_tree.insert("", "end", values=row_values)

        if hasattr(self, "total_summary_lbl"):
            self.total_summary_lbl.config(text=f"  📊 Total Units: {total_qty} | Total: {sym} {sum_unit_prices:,.2f} | Grand Total: {sym} {grand_total:,.2f} | Txns: {count}")

    def generate_bill_text(self):
        shop_title = self.bill_title_entry.get().strip()
        shop_addr = self.bill_addr_entry.get().strip()
        shop_phone = self.bill_phone_entry.get().strip()
        shop_email = self.bill_email_entry.get().strip()
        tax_id = self.bill_taxid_entry.get().strip()
        sym, _ = self.get_current_currency_conversion()
        footer_msg = self.bill_footer_entry.get().strip()
        
        try:
            tax_rate = float(self.bill_tax_entry.get().strip())
        except ValueError:
            tax_rate = 0.0

        try:
            disc_rate = float(self.bill_disc_entry.get().strip())
        except ValueError:
            disc_rate = 0.0

        sales_to_print = []
        if self.print_all_var.get():
            sales_to_print = self.data["sales"]
        else:
            for var, s in self.sales_check_vars.values():
                if var.get():
                    sales_to_print.append(s)
            if not sales_to_print:
                sales_to_print = self.data["sales"]

        lines = []
        lines.append("=============================================")
        lines.append(f"{shop_title:^45}")
        lines.append(f"{shop_addr:^45}")
        lines.append(f"Phone: {shop_phone} | Email: {shop_email}")
        lines.append(f"NTN / STRN Reg: {tax_id}")
        lines.append(f"Date: {datetime.now().strftime('%Y-%m-%d %I:%M %p'):^39}")
        lines.append("=============================================")
        lines.append(f"{'Item Description':<27} | {'Qty':<3} | {f'Price ({sym})':<8}")
        lines.append("---------------------------------------------")

        subtotal = 0.0
        for s in sales_to_print:
            name = s.get("raw_name", s.get("name", "Product"))[:26]
            qty = s.get("qty", 1)
            price = s.get("price", 0.0)
            subtotal += price
            lines.append(f"{name:<27} | {qty:<3} | {sym} {price:<7.2f}")

        disc_amount = subtotal * (disc_rate / 100.0)
        discounted_subtotal = subtotal - disc_amount
        tax_amount = discounted_subtotal * (tax_rate / 100.0)
        grand_total = discounted_subtotal + tax_amount

        lines.append("---------------------------------------------")
        lines.append(f"{'Subtotal:':<33} {sym} {subtotal:>8.2f}")
        if disc_rate > 0:
            lines.append(f"{f'Discount ({disc_rate}%):':<33} -{sym} {disc_amount:>8.2f}")
        if tax_rate > 0:
            lines.append(f"{f'Tax ({tax_rate}%):':<33} {sym} {tax_amount:>8.2f}")
        lines.append(f"{'GRAND TOTAL:':<33} {sym} {grand_total:>8.2f}")
        lines.append("=============================================")
        lines.append(f"{footer_msg:^45}")
        lines.append("=============================================")
        return "\n".join(lines)

    def dispatch_raw_thermal_print(self, custom_text=None):
        if not self.data["sales"]:
            messagebox.showwarning("Warning", "No sales records to print.")
            return

        selected_printer = self.printer_dropdown.var.get()
        bill_str = custom_text if custom_text else self.generate_bill_text()
        
        print_payload = (bill_str + "\n\n\n\x1d\x56\x41\x00").encode("utf-8", errors="ignore")

        try:
            if sys.platform == "win32":
                import win32print
                printer_name = win32print.GetDefaultPrinter()
                hPrinter = win32print.OpenPrinter(printer_name)
                try:
                    hJob = win32print.StartDocPrinter(hPrinter, 1, ("The Duck Till POS Receipt", None, "RAW"))
                    win32print.StartPagePrinter(hPrinter)
                    win32print.WritePrinter(hPrinter, print_payload)
                    win32print.EndPagePrinter(hPrinter)
                    win32print.EndDocPrinter(hPrinter)
                finally:
                    win32print.ClosePrinter(hPrinter)
            else:
                temp_filename = "the_duck_till_receipt_temp.txt"
                with open(temp_filename, "w", encoding="utf-8") as f:
                    f.write(bill_str)
                os.system(f"lpr {temp_filename}")

            messagebox.showinfo("Print Success", f"Receipt successfully transmitted and printed via:\n[{selected_printer}]")
        except Exception as e:
            try:
                temp_filename = "the_duck_till_receipt_temp.txt"
                with open(temp_filename, "w", encoding="utf-8") as f:
                    f.write(bill_str)
                os.startfile(temp_filename, "print")
                messagebox.showinfo("Print Sent", f"Print job dispatched via Windows system print queue for [{selected_printer}]")
            except Exception as ex:
                messagebox.showerror("Print Error", f"Could not transmit print job:\n{str(ex)}")

    def on_return_select(self, event):
        selected = self.return_tree.selection()
        if not selected:
            return
        vals = self.return_tree.item(selected[-1], "values")
        for s in reversed(self.data["sales"]):
            if s["name"] == vals[0] and s["timestamp"] == vals[3]:
                self.return_preview_ref = self.render_image_on_canvas(self.return_preview_canvas, s.get("image", ""), 290, 200)
                break

    def process_return_item(self):
        selected = self.return_tree.selection()
        if not selected:
            messagebox.showwarning("Warning", "Select purchased item(s) to process return.")
            return

        confirm = self.data["settings"].get("confirm_actions", True)
        if confirm and not messagebox.askyesno("Confirm Return", f"Are you sure you want to process returns for the {len(selected)} selected item(s) and restock them?"):
            return

        for item_id in selected:
            vals = self.return_tree.item(item_id, "values")
            target_sale = next((s for s in self.data["sales"] if s["name"] == vals[0] and s["timestamp"] == vals[3]), None)
            if not target_sale:
                continue

            prod_id = target_sale.get("id")
            qty_returned = target_sale.get("qty", 1)

            restocked = False
            for p in self.data["inventory"]:
                if str(p.get("id", "")) == str(prod_id):
                    if p.get("stock", 0) < INFINITY_VAL:
                        p["stock"] += qty_returned
                    restocked = True
                    break

            if not restocked:
                self.data["inventory"].append({
                    "id": prod_id, "name": target_sale.get("raw_name", target_sale["name"]),
                    "category": target_sale["category"], "price": target_sale["price"] / max(1, qty_returned),
                    "stock": qty_returned, "image": target_sale.get("image", "")
                })

            self.data["sales"].remove(target_sale)

        save_data(self.data)
        self.refresh_inventory_list()
        self.refresh_sale_list()
        self.refresh_report_list()
        self.refresh_return_list()
        self.populate_sales_checklist()

        messagebox.showinfo("Return Processed", f"Successfully processed returns for {len(selected)} item(s) and restocked inventory!")

    def refresh_return_list(self, *args):
        if not hasattr(self, "return_tree"):
            return
        sym, _ = self.get_current_currency_conversion()
        for row in self.return_tree.get_children():
            self.return_tree.delete(row)
        query = self.return_search_var.get().lower().strip() if hasattr(self, "return_search_var") else ""
        sort_rule = self.return_sort_dropdown.var.get() if hasattr(self, "return_sort_dropdown") else "Default"

        filtered_sales = []
        for s in reversed(self.data["sales"]):
            if not query or query in s["name"].lower() or query in s["category"].lower() or query in str(s["price"]) or query in s["timestamp"].lower():
                filtered_sales.append(s)

        sorted_sales = self.sort_sales_list(filtered_sales, sort_rule)

        for s in sorted_sales:
            row_values = [s["name"], s["category"], f"{sym} {s['price']:.2f}", s["timestamp"]]
            self.return_tree.insert("", "end", values=row_values)

    def toggle_preferred_path_setting(self):
        val = self.pref_path_var.get()
        if val:
            chosen_file = filedialog.asksaveasfilename(
                defaultextension=".json",
                filetypes=[("JSON Database Files", "*.json")],
                title="Select Preferred Storage Path for Database Backups"
            )
            if chosen_file:
                self.data["settings"]["save_preferred_path"] = True
                self.data["settings"]["preferred_path"] = chosen_file
                save_data(self.data)
                messagebox.showinfo("Path Saved", f"Preferred backup path set to:\n{chosen_file}")
            else:
                self.pref_path_var.set(False)
                self.data["settings"]["save_preferred_path"] = False
        else:
            self.data["settings"]["save_preferred_path"] = False
            self.data["settings"]["preferred_path"] = ""
            save_data(self.data)
            messagebox.showinfo("Disabled", "Preferred path saving disabled.")

    def update_confirm_setting(self, val):
        self.data["settings"]["confirm_actions"] = val
        save_data(self.data)

    def apply_resolution(self):
        res = self.res_dropdown.var.get()
        self.data["settings"]["resolution"] = res
        save_data(self.data)
        if res == "Fullscreen":
            self.root.attributes("-fullscreen", True)
        else:
            self.root.attributes("-fullscreen", False)
            self.root.geometry(res)

    def toggle_maximize(self):
        state = self.root.state()
        if state == "zoomed":
            self.root.state("normal")
        else:
            self.root.state("zoomed")

    def apply_tab_order(self):
        sel = self.tab_order_dropdown.var.get()
        if "Inventory" in sel:
            new_order = ["Inventory", "Point of Sale (Cart)", "Sales Log", "Return", "Bill System", "Settings", "Refresh Controls"]
        elif "Point of Sale" in sel:
            new_order = ["Point of Sale (Cart)", "Inventory", "Sales Log", "Return", "Bill System", "Settings", "Refresh Controls"]
        elif "Sales Log" in sel:
            new_order = ["Sales Log", "Inventory", "Point of Sale (Cart)", "Return", "Bill System", "Settings", "Refresh Controls"]
        elif "Return" in sel:
            new_order = ["Return", "Inventory", "Point of Sale (Cart)", "Sales Log", "Bill System", "Settings", "Refresh Controls"]
        elif "Bill System" in sel:
            new_order = ["Bill System", "Inventory", "Point of Sale (Cart)", "Sales Log", "Return", "Settings", "Refresh Controls"]
        elif "Refresh Controls" in sel:
            new_order = ["Refresh Controls", "Inventory", "Point of Sale (Cart)", "Sales Log", "Return", "Bill System", "Settings"]
        else:
            new_order = ["Settings", "Inventory", "Point of Sale (Cart)", "Sales Log", "Return", "Bill System", "Refresh Controls"]

        self.data["settings"]["tab_order"] = new_order
        save_data(self.data)
        self.reload_ui()
        messagebox.showinfo("Success", "Tab placement updated successfully!")

    def reset_default_theme(self):
        self.data["settings"]["theme_name"] = "Dark Gold"
        self.data["settings"]["custom_colors"] = PRESET_THEMES["Dark Gold"].copy()
        save_data(self.data)
        self.colors = self.data["settings"]["custom_colors"]
        self.reload_ui()
        messagebox.showinfo("Reset", "Theme reverted to Dark Gold default!")

    def apply_preset_theme(self):
        theme_name = self.theme_dropdown.var.get()
        if theme_name in PRESET_THEMES:
            self.data["settings"]["theme_name"] = theme_name
            self.data["settings"]["custom_colors"] = PRESET_THEMES[theme_name].copy()
            save_data(self.data)
            self.colors = self.data["settings"]["custom_colors"]
            self.reload_ui()

    def pick_color(self, key_name):
        color_code = colorchooser.askcolor(title=f"Choose {key_name} color")[1]
        if color_code:
            self.data["settings"]["theme_name"] = "Custom"
            self.data["settings"]["custom_colors"][key_name] = color_code
            save_data(self.data)
            self.colors = self.data["settings"]["custom_colors"]
            self.reload_ui()

    def reload_ui(self):
        self.setup_styles()
        for widget in self.root.winfo_children():
            widget.destroy()
        self.build_ui()
        self.bind_global_shortcuts()

    def toggle_sales_lock(self):
        pin = simpledialog.askstring("Admin Verification", "Enter Admin PIN:", show="*")
        if pin == self.data["settings"].get("admin_pin", "1234"):
            self.data["settings"]["allow_staff_delete_sales"] = not self.lock_sales_var.get()
            save_data(self.data)
            messagebox.showinfo("Updated", "Security rule updated.")
        else:
            self.lock_sales_var.set(not self.data["settings"].get("allow_staff_delete_sales", True))
            messagebox.showerror("Denied", "Incorrect PIN!")

    def change_admin_pin(self):
        old_pin = simpledialog.askstring("Verify PIN", "Enter Current Admin PIN:", show="*")
        if old_pin == self.data["settings"].get("admin_pin", "1234"):
            new_pin = simpledialog.askstring("New PIN", "Enter New Admin PIN:", show="*")
            if new_pin and len(new_pin.strip()) > 0:
                self.data["settings"]["admin_pin"] = new_pin.strip()
                save_data(self.data)
                messagebox.showinfo("Success", "Admin PIN updated successfully.")
        else:
            messagebox.showerror("Denied", "Incorrect PIN.")
